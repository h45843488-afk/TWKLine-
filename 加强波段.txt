import pandas as pd
import numpy as np

def calculate_swing_indicators(df):
    """
    精準轉換通達信/XS邏輯至 Python (Pandas)
    必要欄位：df['close'], df['high'], df['low']
    """
    # 複製一份資料避免改動原始 DataFrame
    data = df.copy()
    
    # ===== 1. 波段拐點 (13, 6 穩定版) =====
    gup6 = (2 * data['close'] + data['high'] + data['low']) / 4
    gup7 = data['low'].rolling(window=13).min()
    gup8 = data['high'].rolling(window=13).max()
    
    # 分母防呆，避免除以零
    denom_gup = (gup8 - gup7).replace(0, np.nan)
    gup9_raw = (gup6 - gup7) / denom_gup * 100
    
    # EMA(..., 13) -> 通達信 adjust=False 衰減模式
    gup9 = gup9_raw.ewm(span=13, adjust=False).mean()
    
    # REF(GUP9, 2) 取前第2天數值
    gup9_ref2 = gup9.shift(2)
    
    # EMA(0.382 * REF(GUP9, 2) + 0.618 * GUP9, 6)
    weighted_gup9 = 0.382 * gup9_ref2 + 0.618 * gup9
    data['波段拐點'] = (weighted_gup9.ewm(span=6, adjust=False).mean() - 50) / 100
    
    # 波段拐點狀態：1=變紅(上升/買點), -1=變青(下降/賣點)
    data['波段拐點_方向'] = np.where(
        data['波段拐點'] > data['波段拐點'].shift(1), 1, 
        np.where(data['波段拐點'] < data['波段拐點'].shift(1), -1, 0)
    )

    # ===== 2. 中期安全線 (55 週期趨勢過濾) =====
    llv60 = data['low'].rolling(window=60).min()
    hhv60 = data['high'].rolling(window=60).max()
    
    denom_price = (hhv60 - llv60).replace(0, np.nan)
    gup_price = (data['close'] - llv60) / denom_price
    
    # MA(EMA(GUP_PRICE, 3), 3)
    ema_gup_price = gup_price.ewm(span=3, adjust=False).mean()
    gup1 = ema_gup_price.rolling(window=3).mean()
    
    # EMA(GUP1 - 0.5, 55)
    data['中期安全線'] = (gup1 - 0.5).ewm(span=55, adjust=False).mean()
    
    # 中期安全線狀態：1=洋紅(安全區/看多), 0=灰色(危險區/觀望)
    data['中期安全線_安全區'] = np.where(data['中期安全線'] > data['中期安全線'].shift(1), 1, 0)

    return data
